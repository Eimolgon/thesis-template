% ============================================================
% COLOURS
% ============================================================

\usepackage{xcolor}

% Default chapter colour
\definecolor{chapterblue}{RGB}{0,70,140}


% ============================================================
% HEADER AND FOOTER
% ============================================================

\usepackage{fancyhdr}

% Default chapter colour
\definecolor{chapterblue}{RGB}{0,70,140}

% Blue line thickness
\newcommand{\headerlinethickness}{0.6pt}

% Animation area
\newlength{\headeranimationheight}
\setlength{\headeranimationheight}{10mm}

% Header height
\setlength{\headheight}{18mm}


% ============================================================
% PAGE STYLE
% ============================================================

\pagestyle{fancy}

\fancyhf{}

% Page number
\fancyfoot[RO,LE]{\thepage}


% ============================================================
% RIGHT-HAND PAGES
% ============================================================

% Animation only
\fancyhead[RO]{%
    \begin{minipage}[b][\headeranimationheight][b]{0.45\textwidth}
        \raggedleft
        \headeranimation
    \end{minipage}%
}


% ============================================================
% LEFT-HAND PAGES
% ============================================================

% Chapter/section information
\fancyhead[LE]{%
    \small
    \nouppercase{\leftmark}%
}


% ============================================================
% SINGLE LINE BELOW THE HEADER
% ============================================================

% Normal fancyhdr rule
\renewcommand{\headrulewidth}{\headerlinethickness}

% Change its colour
\makeatletter
\renewcommand{\headrule}{%
    \hbox to\headwidth{%
        \color{chapterblue}%
        \leaders\hrule height\headerlinethickness\hfill
    }%
}
\makeatother


% ============================================================
% CHAPTER OPENING PAGES
% ============================================================

\fancypagestyle{plain}{
    \fancyhf{}

    \fancyfoot[RO,LE]{\thepage}

    % Right-hand page: animation
    \fancyhead[RO]{%
        \begin{minipage}[b][\headeranimationheight][b]{0.45\textwidth}
            \raggedleft
            \headeranimation
        \end{minipage}%
    }

    % Left-hand page: chapter/section information
    \fancyhead[LE]{%
        \small
        \nouppercase{\leftmark}%
    }

    % One single line
    \renewcommand{\headrulewidth}{\headerlinethickness}
}


% ============================================================
% ANIMATION IMAGE PLACEHOLDER
% ============================================================

\newcommand{\headeranimation}{%
    % Empty by default
    %
    % Example:
    % \includegraphics[height=9mm]{figures/header/frame-001.pdf}
}


% ============================================================
% CHAPTER COLOUR COMMAND
% ============================================================

\newcommand{\chaptercolor}[1]{%
    \colorlet{chapterblue}{#1}%
}


% ============================================================
% CHAPTER MARKS
% ============================================================

\renewcommand{\chaptermark}[1]{%
    \markboth{%
        \thechapter\quad #1
    }{}
}

\renewcommand{\sectionmark}[1]{%
    \markright{%
        \thesection\quad #1
    }%
}
